cmake_minimum_required(VERSION 3.25)
project(PulseNativeArchiveDialog LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(pulse_archive_dialog SHARED plugin.cpp)
target_include_directories(pulse_archive_dialog PRIVATE "${PULSE_SDK_INCLUDE}")
target_compile_features(pulse_archive_dialog PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(pulse_archive_dialog PRIVATE /W4 /permissive- /utf-8)
    set_target_properties(pulse_archive_dialog PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
