cmake_minimum_required(VERSION 3.25)
project(PulseProjectCapsules LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(com_pulse_project_capsules SHARED plugin.cpp)
target_include_directories(com_pulse_project_capsules PRIVATE "${PULSE_SDK_INCLUDE}" "${CMAKE_CURRENT_BINARY_DIR}")
target_compile_features(com_pulse_project_capsules PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(com_pulse_project_capsules PRIVATE /W4 /permissive- /utf-8 /EHsc)
    set_target_properties(com_pulse_project_capsules PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
