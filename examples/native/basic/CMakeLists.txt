cmake_minimum_required(VERSION 3.25)
project(PulseNativeExample LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(pulse_native_example SHARED plugin.cpp)
target_include_directories(pulse_native_example PRIVATE "${PULSE_SDK_INCLUDE}")
target_compile_features(pulse_native_example PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(pulse_native_example PRIVATE /W4 /permissive- /utf-8)
endif()
