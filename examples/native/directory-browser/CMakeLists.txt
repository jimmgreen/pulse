cmake_minimum_required(VERSION 3.25)
project(PulseDirectoryBrowserExample LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(pulse_directory_browser SHARED plugin.cpp)
add_executable(pulse_directory_browser_test provider_test.cpp plugin.cpp)
foreach(target pulse_directory_browser pulse_directory_browser_test)
    target_include_directories(${target} PRIVATE "${PULSE_SDK_INCLUDE}")
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8)
        set_target_properties(${target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endforeach()
