cmake_minimum_required(VERSION 3.25)
project(PulseNativeDirectoryTransfer LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(pulse_directory_transfer SHARED plugin.cpp)
target_include_directories(pulse_directory_transfer PRIVATE "${PULSE_SDK_INCLUDE}")
target_compile_features(pulse_directory_transfer PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(pulse_directory_transfer PRIVATE /W4 /permissive- /utf-8)
    set_target_properties(pulse_directory_transfer PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
