cmake_minimum_required(VERSION 3.25)
project(PulseNativeInspector LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(pulse_inspector SHARED plugin.cpp)
target_include_directories(pulse_inspector PRIVATE "${PULSE_SDK_INCLUDE}")
target_compile_features(pulse_inspector PRIVATE cxx_std_17)
if(MSVC)
    target_compile_options(pulse_inspector PRIVATE /W4 /permissive- /utf-8)
    set_target_properties(pulse_inspector PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
