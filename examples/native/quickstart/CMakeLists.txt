cmake_minimum_required(VERSION 3.25)
project(PulseNativeQuickstart LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(pulse_quickstart SHARED plugin.cpp)
target_include_directories(pulse_quickstart PRIVATE "${PULSE_SDK_INCLUDE}")
target_compile_features(pulse_quickstart PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(pulse_quickstart PRIVATE /W4 /permissive- /utf-8 /EHsc)
    set_target_properties(pulse_quickstart PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
