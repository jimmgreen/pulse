cmake_minimum_required(VERSION 3.25)
project(PulseReactiveExample LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
add_library(pulse_reactive SHARED plugin.cpp)
target_include_directories(pulse_reactive PRIVATE "${PULSE_SDK_INCLUDE}")
target_compile_features(pulse_reactive PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(pulse_reactive PRIVATE /W4 /permissive- /utf-8)
    set_target_properties(pulse_reactive PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_executable(pulse_client_wrapper_test wrapper_test.cpp)
target_include_directories(pulse_client_wrapper_test PRIVATE "${PULSE_SDK_INCLUDE}")
target_compile_features(pulse_client_wrapper_test PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(pulse_client_wrapper_test PRIVATE /W4 /permissive- /utf-8)
endif()
