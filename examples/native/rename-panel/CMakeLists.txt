cmake_minimum_required(VERSION 3.25)
project(PulseNativeRenameExample LANGUAGES CXX)
set(PULSE_SDK_INCLUDE "${CMAKE_CURRENT_LIST_DIR}/../../../sdk/native/include" CACHE PATH "Public Pulse SDK include directory")
foreach(variant IN ITEMS lower upper)
    add_library(pulse_rename_${variant} SHARED plugin.cpp)
    target_include_directories(pulse_rename_${variant} PRIVATE "${PULSE_SDK_INCLUDE}")
    target_compile_features(pulse_rename_${variant} PRIVATE cxx_std_17)
    if(variant STREQUAL "upper")
        target_compile_definitions(pulse_rename_${variant} PRIVATE PULSE_SAMPLE_UPPER_CASE=1)
    endif()
    if(MSVC)
        target_compile_options(pulse_rename_${variant} PRIVATE /W4 /permissive- /utf-8)
        set_target_properties(pulse_rename_${variant} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
endforeach()
