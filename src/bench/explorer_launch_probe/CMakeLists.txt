cmake_minimum_required(VERSION 3.25)
project(PulseExplorerLaunchProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_compile_definitions(UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_compile_options(/W4 /WX /permissive- /utf-8 /EHsc)
include_directories("${ROOT}/src" "${ROOT}/third_party/detours/src" "${ROOT}/build/generated")
add_library(probe_route STATIC route.cpp)
target_link_libraries(probe_route PUBLIC shell32)
add_library(explorer_probe MODULE addon.cpp)
set_target_properties(explorer_probe PROPERTIES PREFIX "" SUFFIX ".node")
target_link_libraries(explorer_probe PRIVATE probe_route "${ROOT}/build/pulse_detours.lib")
add_executable(explorer_bridge WIN32 bridge.cpp "${ROOT}/src/app/open_request_pipe.cpp")
target_link_libraries(explorer_bridge PRIVATE probe_route ole32 advapi32 user32)
add_executable(explorer_receiver receiver.cpp "${ROOT}/src/app/single_instance_coordinator.cpp"
    "${ROOT}/src/app/open_request_pipe.cpp" "${ROOT}/src/app/this_pc_argument.cpp"
    "${ROOT}/src/common/runtime_log.cpp")
target_link_libraries(explorer_receiver PRIVATE ole32 advapi32 user32 psapi)
add_executable(explorer_route_test route_test.cpp)
target_link_libraries(explorer_route_test PRIVATE probe_route)
add_executable(explorer_watch watch.cpp)
target_link_libraries(explorer_watch PRIVATE user32)
